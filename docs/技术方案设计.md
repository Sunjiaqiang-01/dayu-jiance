## 二、API接口规范

### 2.1 接口总览

| 接口 | 方法 | 路径 | 说明 |
|------|------|------|------|
| 健康检查 | GET | `/api/v1/health` | 服务状态 |
| 创建检测 | POST | `/api/v1/detection` | 提交检测任务 |
| 查询状态 | GET | `/api/v1/detection/{id}` | 获取检测进度 |
| 获取结果 | GET | `/api/v1/detection/{id}/result` | 获取完整结果 |
| 生成报告 | POST | `/api/v1/report` | 创建报告任务 |
| 查询报告 | GET | `/api/v1/report/{id}` | 获取报告状态 |
| 下载报告 | GET | `/api/v1/report/{id}/download` | 下载PDF文件 |
| 获取行业列表 | GET | `/api/v1/industries` | 获取可选行业 |
| 获取场景列表 | GET | `/api/v1/industries/{industry}/scenes` | 获取行业场景 |

### 3.2 数据表定义

#### 3.2.1 detection_record (检测记录表)

```sql
CREATE TABLE detection_record (
    id VARCHAR(36) PRIMARY KEY,
    brand_name VARCHAR(255) NOT NULL,
    industry VARCHAR(100) NOT NULL,
    status VARCHAR(50) NOT NULL DEFAULT 'pending',
    
    -- 进度信息
    current_step INT DEFAULT 1,
    progress INT DEFAULT 0,
    
    -- 时间戳
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    completed_at TIMESTAMP,
    
    -- 汇总结果
    overall_score INT,
    visibility_level VARCHAR(50),
    
    -- 完整JSON结果
    result_json TEXT,
    
    -- 元数据
    total_duration INT,
    error_message TEXT,
    
    INDEX idx_brand_name (brand_name),
    INDEX idx_industry (industry),
    INDEX idx_status (status),
    INDEX idx_created_at (created_at)
);
```

CREATE TABLE detection_step (

### 4.2 平台抽象层

```python
# backend/app/services/ai_platform/base.py
from abc import ABC, abstractmethod
from dataclasses import dataclass
from typing import List, Optional, Dict, Any

@dataclass
class AIResponse:
    success: bool
    content: str
    platform: str
    model: str
    tokens_used: Optional[int] = None
    latency_ms: Optional[int] = None
    error: Optional[str] = None

@dataclass  
class PlatformStatus:
    name: str
    available: bool
    latency_ms: Optional[int] = None
    error: Optional[str] = None

class BaseAIPlatform(ABC):
    def __init__(self, api_key: str, base_url: str, model: str):
        self.api_key = api_key
        self.base_url = base_url
        self.model = model
    
    @abstractmethod
    async def chat(self, messages: List[Dict], **kwargs) -> AIResponse:
        """发送对话请求"""
        pass
    
    @abstractmethod
    async def health_check(self) -> PlatformStatus:
        """健康检查"""
        pass
```

### 4.3 平台池实现

```python
# backend/app/services/ai_platform/pool.py
from typing import List, Optional
from .base import BaseAIPlatform, AIResponse, PlatformStatus

class AIPlatformPool:
    def __init__(self):
        self.platforms: Dict[str, BaseAIPlatform] = {}
        self.fallback_order: List[str] = []
    
    def register(self, name: str, platform: BaseAIPlatform, priority: int = 0):
        """注册AI平台"""
        self.platforms[name] = platform
        # 按优先级排序
        self.fallback_order.sort(key=lambda p: self.platforms[p].priority)
    
    async def chat_with_fallback(
        self, 
        messages: List[Dict],
        preferred_platforms: Optional[List[str]] = None,
        timeout: int = 30
    ) -> AIResponse:
        """带降级的聊天请求"""
        platforms_to_try = preferred_platforms or self.fallback_order
        
        for platform_name in platforms_to_try:
            platform = self.platforms.get(platform_name)
            if not platform:
                continue
            
            try:
                response = await asyncio.wait_for(
                    platform.chat(messages),
                    timeout=timeout
                )
                if response.success:
                    return response
            except Exception as e:
                continue
        
        return AIResponse(
            success=False,
            content="",
            platform="",
            model="",
            error="所有平台均不可用"
        )
    
    async def health_check_all(self) -> Dict[str, PlatformStatus]:
        """检查所有平台状态"""
        tasks = [
            platform.health_check()
            for platform in self.platforms.values()
        ]
        results = await asyncio.gather(*tasks, return_exceptions=True)
        
        return {
            name: result if isinstance(result, PlatformStatus) else PlatformStatus(
                name=name,
                available=False,
                error=str(result)
            )
            for name, result in zip(self.platforms.keys(), results)
        }
```

### 5.4 生成服务

```python
# backend/app/services/report_service.py
import asyncio
from datetime import datetime
from pathlib import Path
from typing import Optional
import weasyprint
from jinja2 import Environment, FileSystemLoader

class ReportService:
    def __init__(self, template_dir: Path, output_dir: Path):
        self.template_dir = template_dir
        self.output_dir = output_dir
        self.env = Environment(loader=FileSystemLoader(template_dir))
    
    async def generate(
        self,
        detection_id: str,
        brand_name: str,
        industry: str,
        result_data: dict
    ) -> str:
        """生成PDF报告"""
        # 准备模板数据
        template_data = {
            "brand_name": brand_name,
            "industry": industry,
            "report_date": datetime.now().strftime("%Y年%m月%d日"),
            "overall_score": result_data.get("summary", {}).get("overall_score", 0),
            "visibility_level": result_data.get("summary", {}).get("visibility_level", "未知"),
            "scene_coverage": result_data.get("scene_coverage", {}),
            "competitors": result_data.get("competitors", {}),
            "opportunities": result_data.get("opportunities", {}),
            "action_items": result_data.get("action_items", {}),
        }
        
        # 渲染HTML
        template = self.env.get_template("report/report.html")
        html_content = template.render(**template_data)
        
        # 生成PDF
        output_path = self.output_dir / f"{brand_name}_AI品牌机会发现报告_{datetime.now().strftime('%Y%m%d')}.pdf"
        
        await asyncio.to_thread(
            weasyprint.HTML(string=html_content).write_pdf,
            output_path
        )
        
        return str(output_path)
```
